// These fixtures model external receipts to test rejection behavior. They do

func documentationModelJSON(t *testing.T, value any) []byte {
	t.Helper()
	var data, err = json.MarshalIndent(value, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	return append(data, '\n')
}

func documentationModelGit(t *testing.T, root string, arguments ...string) string {
	t.Helper()
	var command = exec.Command("git", append([]string{"-C", root}, arguments...)...)
	command.Env = append(os.Environ(), "GIT_AUTHOR_NAME=Synthetic D03 control", "GIT_AUTHOR_EMAIL=synthetic@example.invalid",
		"GIT_COMMITTER_NAME=Synthetic D03 control", "GIT_COMMITTER_EMAIL=synthetic@example.invalid", "GIT_EDITOR=true")
	var output, err = command.CombinedOutput()
	if err != nil {
		t.Fatalf("synthetic Git fixture: %v %s", err, output)
	}
	return strings.TrimSpace(string(output))
}

func documentationModelWrite(t *testing.T, name string, data []byte) {
	t.Helper()
	if err := os.MkdirAll(filepath.Dir(name), 0755); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(name, data, 0644); err != nil {
		t.Fatal(err)
	}
}

func documentationCopyBytes(original map[string][]byte) map[string][]byte {
	var output = make(map[string][]byte)
	for key, value := range original {
		output[key] = append([]byte{}, value...)
	}
	return output
}

func documentationMaterializeRecord(t *testing.T, original EvidenceRecord, data map[string][]byte, runID string) EvidenceRecord {
	t.Helper()
	var record EvidenceRecord
	if err := json.Unmarshal(documentationModelJSON(t, original), &record); err != nil {
		t.Fatal(err)
	}
	record.Artifacts = nil
	var directory = t.TempDir()
	for name, content := range data {
		var path = filepath.Join(directory, name)
		documentationModelWrite(t, path, content)
		record.Artifacts = append(record.Artifacts, EvidenceArtifact{Path: path, SHA256: nativeEncodingDigest(content), RunID: runID})
	}
	return record
}

func documentationMutateJSON(t *testing.T, data map[string][]byte, name string, change func(map[string]any)) {
	t.Helper()
	var value map[string]any
	if err := json.Unmarshal(data[name], &value); err != nil {
		t.Fatal(err)
	}
	change(value)
	data[name] = documentationModelJSON(t, value)
}

func documentationRebindAppLog(t *testing.T, data map[string][]byte) {
	t.Helper()
	documentationMutateJSON(t, data, "ios-test-diagnostics.json", func(value map[string]any) {
		value["sanitizedLogSha256"] = nativeEncodingDigest(data["xcodetest.log"])
	})
}

func TestDocumentationFormalScopeRequiresBothConsumers(t *testing.T) {
	var model = newDocumentationIntakeModel(t)
	var app = documentationMaterializeRecord(t, model.app, model.appBytes, "123456")
	var consumer = documentationMaterializeRecord(t, model.consumer, model.docBytes, "")
	model.manifest.Records = []EvidenceRecord{app, consumer}
	var report, err = ValidateEvidenceManifest(&model.manifest, model.contracts, "local")
	if err != nil || report.PassedCases != 1 || report.ClosureComplete {
		t.Fatalf("synthetic complete intake control: %+v %v", report, err)
	}
	for _, row := range []EvidenceRecord{app, consumer} {
		var changed = model.manifest
		changed.Records = []EvidenceRecord{row}
		var partial, _ = ValidateEvidenceManifest(&changed, model.contracts, "local")
		if partial == nil || partial.PassedCases != 0 || partial.ClosureComplete {
			t.Fatal("one consumer alone closed D03")
		}
	}
	for _, mutation := range []string{"wrong_producer", "wrong_scope", "missing_artifact", "summary_only"} {
		t.Run(mutation, func(t *testing.T) {
			var changed EvidenceManifest
			_ = json.Unmarshal(documentationModelJSON(t, model.manifest), &changed)
			switch mutation {
			case "wrong_producer":
				changed.Records[0].TestID = "tools/language-acceptance:runDocs"
			case "wrong_scope":
				changed.Records[0].Requirements = []string{"D03", "RP46"}
			case "missing_artifact":
				changed.Records[0].Artifacts = changed.Records[0].Artifacts[1:]
			case "summary_only":
				changed.Records[1].Artifacts = nil
			}
			var invalid, invalidErr = ValidateEvidenceManifest(&changed, model.contracts, "local")
			if invalidErr == nil || invalid.PassedCases != 0 || invalid.ClosureComplete {
				t.Fatal("invalid D03 evidence survived the actual manifest validator")
			}
		})
	}
}
