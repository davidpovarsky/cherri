// unsupported or malformed field. They do not stand in for Apple execution.
func p10DecodeJSONDictionary(raw any, resolved map[string]any) (map[string]any, error) {
 outer, ok := raw.(map[string]any)
 if !ok || outer["WFSerializationType"] != "WFDictionaryFieldValue" { return nil, fmt.Errorf("invalid Dictionary wrapper") }
 value, ok := outer["Value"].(map[string]any)
 if !ok { return nil, fmt.Errorf("missing Dictionary value") }
 rows, ok := value["WFDictionaryFieldValueItems"].([]any)
 if !ok { return nil, fmt.Errorf("missing explicit Dictionary item list") }
 result := make(map[string]any,len(rows))
 for i, rawRow := range rows {
  row, ok := rawRow.(map[string]any)
  if !ok { return nil, fmt.Errorf("invalid Dictionary row %d",i) }
  key, err := p10DecodeJSONText(row["WFKey"],resolved)
  if err != nil { return nil,fmt.Errorf("Dictionary key %d: %w",i,err) }
  if _, duplicate := result[key]; duplicate { return nil,fmt.Errorf("duplicate Dictionary key %q",key) }
  item, err := p10DecodeJSONItem(row,resolved)
  if err != nil { return nil,fmt.Errorf("Dictionary field %q: %w",key,err) }
  result[key]=item
 }
 return result,nil
}

func p10DecodeJSONReference(ref map[string]any, resolved map[string]any) (any,error) {
 kind, _ := ref["Type"].(string)
 key := ""

 case "ActionOutput":
  id, _ := ref["OutputUUID"].(string)
  if id != "" { key = kind+":"+id }
 case "Variable":

}

func p10DecodeJSONText(raw any, resolved map[string]any) (string,error) {

func p10DecodeJSONItem(row map[string]any, resolved map[string]any) (any,error) {
 kind := uint64(99)

 case uint64:kind=raw

   return p10DecodeJSONReference(ref,resolved)

func p10CloneJSONWire(raw any) any {
 switch value:=raw.(type) {
 case map[string]any:

 case []any:
  copy:=make([]any,len(value));for i,v:=range value { copy[i]=p10CloneJSONWire(v) };return copy
 default:return raw
 }
}
